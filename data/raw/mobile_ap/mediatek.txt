    const heading = drop.querySelector("h3");
    const list = drop.querySelector("ul");
    const items = drop.querySelectorAll("ul li a");

    if (!heading || !list) return;

    // Detect text container
    const textSpan =
      heading.querySelector(".heading-text") ||
      heading.querySelector(".head_text");

    // Toggle dropdown
    heading.addEventListener("click", function () {

      const isOpen = list.classList.contains("active");

      // Close both dropdowns first
      closeAll();

      // If clicked dropdown was previously closed, open it
      if (!isOpen) {

        // Add active class to the UL
        list.classList.add("active");

        // Add menu-open class to the clicked H3
        heading.classList.add("menu-open");
      }
    });

    // Item click
    items.forEach(item => {
      item.addEventListener("click", function (e) {

        // Prevent default only for category dropdown
        if (drop.querySelector(".block")) {
          e.preventDefault();
        }

        const text = this.textContent.trim();

        // Update only the text
        if (textSpan) {
          textSpan.textContent = text;
        }

        // Close dropdowns
        closeAll();
      });
    });

  });

});
